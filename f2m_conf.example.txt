# ---------------------------------------------
# f2m_conf.txt 設定例
# ---------------------------------------------
# このファイルを f2m_conf.txt にコピーし、設置先に合わせて値を変更します。
# 値の後ろにインラインコメントは書かず、コメントは独立行に記述します。

# ---------------------------------------------
# 基本設定
# ---------------------------------------------
# フォームHTMLの hidden F2M_ID と一致させます。
F2M_ID : contact

# メール送信元、管理者宛先、管理者宛メール件名です。
F2M_FROM : no-reply@example.com
F2M_TO : admin@example.com
F2M_SUBJECT : お問い合わせがありました

# ---------------------------------------------
# 入力チェック
# ---------------------------------------------
# F2M_CHK は必須項目です。
# F2M_JPNAME はフォーム項目名と表示名の対応表です。
# F2M_CHK_EMAIL はメール形式チェック対象です。
# F2M_CHK_EQ は一致チェック対象です。
F2M_CHK : name,mail,mail2,contact
F2M_JPNAME : name:お名前,mail:メールアドレス,mail2:メールアドレス確認用,contact:お問い合わせ内容,attach:添付ファイル
F2M_CHK_EMAIL : mail
F2M_CHK_EQ : mail:mail2

# ---------------------------------------------
# SPAM対策
# ---------------------------------------------
# F2M_HONEYPOT_FLD は通常利用者に表示しないフォーム項目です。
# F2M_MIN_SUBMIT_SECONDS は受付から送信までの最短秒数です。
F2M_HONEYPOT_FLD : website
F2M_MIN_SUBMIT_SECONDS : 2

# ---------------------------------------------
# 自動返信
# ---------------------------------------------
# F2M_RESV_TO_FLD に指定したフォーム項目を自動返信先として使用します。
F2M_RESV_TO_FLD : mail
F2M_RESV_SUBJECT : お問い合わせありがとうございます
F2M_RESV_TMPL : ./php/templates/resv.tpl

# ---------------------------------------------
# テンプレート
# ---------------------------------------------
# 入力画面、確認画面、完了画面、管理者宛メール本文テンプレートです。
F2M_FORM : ./form.html
F2M_CONFIRM : ./php/templates/confirm.html
F2M_THANKS : ./php/templates/thx.html
F2M_MAIL_TMPL : ./php/templates/mail.tpl

# 旧版互換用です。通常の入力エラーは F2M_FORM 内に表示されます。
# F2M_FORMERR : ./php/templates/err.html

# ---------------------------------------------
# 保存
# ---------------------------------------------
# CSV保存先はWebから直接参照できない場所を推奨します。
F2M_CSV : ./storage/csv/contact.csv

# 現行実装ではCSV出力はSJIS固定です。互換用の設定値です。
F2M_CSV_CHARSET : SJIS

# ---------------------------------------------
# メール送信
# ---------------------------------------------
# SMTPを使用する場合の設定例です。
F2M_MAIL_SENDER : smtp
F2M_SMTP_HOST : smtp.example.com
F2M_SMTP_PORT : 587

# SMTP認証を使用する場合のみ有効化します。
# F2M_SMTP_AUTH : true
# F2M_SMTP_USER : smtp-user@example.com
# F2M_SMTP_PASSWORD : smtp-password

# ローカルのMailpitでテストする場合の例です。
# F2M_MAIL_SENDER : smtp
# F2M_SMTP_HOST : 127.0.0.1
# F2M_SMTP_PORT : 1025

# ---------------------------------------------
# エラー表示
# ---------------------------------------------
# 本番環境では false を指定します。
F2M_DEBUG : false

# ---------------------------------------------
# 添付ファイル
# ---------------------------------------------
# 添付ファイルを使用しない場合はコメントアウトします。
F2M_ATTACH_FLD : attach
F2M_ATTACH_MAX : 3M
